  <script>
    class WebAudioEngine {
      constructor() {
        this.ctx = null;
        this.masterGain = null;
        this.analyser = null;
        this.kickVol = 0.9;
        this.snareVol = 0.8;
        this.hiHatVol = 0.7;
      }

      init() {
        if (this.ctx) return;
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        this.ctx = new AudioCtx();
        this.masterGain = this.ctx.createGain();
        this.masterGain.gain.value = 0.8;

        this.analyser = this.ctx.createAnalyser();
        this.analyser.fftSize = 256;

        this.masterGain.connect(this.analyser);
        this.analyser.connect(this.ctx.destination);
      }

      resume() {
        if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume();
      }

      setKickVol(v) { this.kickVol = Number(v); }
      setSnareVol(v) { this.snareVol = Number(v); }
      setHiHatVol(v) { this.hiHatVol = Number(v); }

      playKick() {
        this.init(); this.resume();
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(140, now);
        osc.frequency.exponentialRampToValueAtTime(38, now + 0.08);

        gain.gain.setValueAtTime(this.kickVol * 1.2, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.35);

        osc.connect(gain);
        gain.connect(this.masterGain);
        osc.start(now);
        osc.stop(now + 0.35);
      }

      playSnare() {
        this.init(); this.resume();
        const now = this.ctx.currentTime;
        const bufferSize = this.ctx.sampleRate * 0.15;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;
        const filter = this.ctx.createBiquadFilter();
        filter.type = 'highpass';
        filter.frequency.value = 1000;

        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(this.snareVol, now);
        gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.masterGain);
        noise.start(now);
        noise.stop(now + 0.15);
      }

      playHiHat() {
        this.init(); this.resume();
        const now = this.ctx.currentTime;
        const bufferSize = this.ctx.sampleRate * 0.05;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;
        const filter = this.ctx.createBiquadFilter();
        filter.type = 'highpass';
        filter.frequency.value = 7000;

        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(this.hiHatVol, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.05);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.masterGain);
        noise.start(now);
        noise.stop(now + 0.05);
      }
    }

    const audioEngine = new WebAudioEngine();

    let isPlaying = false;
    let bpm = 120;
    let currentStep = 0;
    let timerId = null;

    const tracks = [
      { id: 'kick', name: 'Kick 808', color: '#f97316', steps: [true, false, false, false, true, false, false, false, true, false, false, false, true, false, false, false] },
      { id: 'snare', name: 'Snare Clap', color: '#f59e0b', steps: [false, false, false, false, true, false, false, false, false, false, false, false, true, false, false, false] },
      { id: 'hihat', name: 'HiHat Trap', color: '#eab308', steps: [true, true, true, true, true, true, true, true, true, true, true, true, true, true, true, true] }
    ];

    function renderSequencer() {
      const grid = document.getElementById('sequencerGrid');
      grid.innerHTML = '';

      tracks.forEach((track) => {
        const row = document.createElement('div');
        row.className = 'flex items-center gap-3 p-3 bg-neutral-950 border border-neutral-800 rounded-xl';

        const title = document.createElement('span');
        title.className = 'text-xs font-bold w-24 truncate';
        title.style.color = track.color;
        title.innerText = track.name;
        row.appendChild(title);

        const stepsDiv = document.createElement('div');
        stepsDiv.className = 'grid grid-cols-16 gap-1 flex-1';

        track.steps.forEach((active, idx) => {
          const btn = document.createElement('button');
          btn.id = `step-${track.id}-${idx}`;
          btn.className = `h-10 rounded-lg font-mono text-[10px] font-bold transition ${active ? 'active-step' : 'bg-neutral-900 border border-neutral-800 text-neutral-600'}`;
          btn.innerText = idx + 1;
          btn.onclick = () => {
            track.steps[idx] = !track.steps[idx];
            renderSequencer();
          };
          stepsDiv.appendChild(btn);
        });

        row.appendChild(stepsDiv);
        grid.appendChild(row);
      });
    }

    function togglePlay() {
      isPlaying = !isPlaying;
      const playBtn = document.getElementById('playBtn');
      if (isPlaying) {
        playBtn.innerText = '⏹ إيقاف التشغيل';
        const intervalMs = (60 / bpm / 4) * 1000;
        timerId = setInterval(runStep, intervalMs);
      } else {
        playBtn.innerText = '▶ تشغيل الإيقاع';
        clearInterval(timerId);
        currentStep = 0;
      }
    }

    function runStep() {
      tracks.forEach((track) => {
        if (track.steps[currentStep]) {
          if (track.id === 'kick') audioEngine.playKick();
          if (track.id === 'snare') audioEngine.playSnare();
          if (track.id === 'hihat') audioEngine.playHiHat();
        }
      });

      document.querySelectorAll('.current-step').forEach(e => e.classList.remove('current-step'));
      tracks.forEach((track) => {
        const el = document.getElementById(`step-${track.id}-${currentStep}`);
        if (el) el.classList.add('current-step');
      });

      currentStep = (currentStep + 1) % 16;
    }

    function updateBpm(val) {
      bpm = Number(val);
      if (isPlaying) {
        clearInterval(timerId);
        timerId = setInterval(runStep, (60 / bpm / 4) * 1000);
      }
    }

    function startVisualizer() {
      const canvas = document.getElementById('spectrumCanvas');
      const ctx = canvas.getContext('2d');

      function draw() {
        requestAnimationFrame(draw);
        ctx.fillStyle = '#0a0a0a';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        if (audioEngine.analyser) {
          const bufferLength = audioEngine.analyser.frequencyBinCount;
          const dataArray = new Uint8Array(bufferLength);
          audioEngine.analyser.getByteFrequencyData(dataArray);

          const barWidth = (canvas.width / bufferLength) * 2;
          let x = 0;
          for (let i = 0; i < bufferLength; i++) {
            const barHeight = (dataArray[i] / 255) * canvas.height;
            ctx.fillStyle = '#f97316';
            ctx.fillRect(x, canvas.height - barHeight, barWidth - 1, barHeight);
            x += barWidth;
          }
        }
      }
      draw();
    }
